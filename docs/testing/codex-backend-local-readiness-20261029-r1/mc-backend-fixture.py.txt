import os, pathlib, subprocess, secrets, json, datetime, time, sys, collections
ROOT=pathlib.Path('/Users/nguyenvanlan/Documents/code/pet-projects/matter-control')
OUT=ROOT/'docs/testing/codex-backend-local-readiness-20261029-r1'
PRIVATE=pathlib.Path('/private/tmp/mc-codex-backend-local-20261029-r2')
def utc(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def run(args,env=None,cwd=ROOT,timeout=600):
    start=utc(); p=subprocess.run(args,cwd=cwd,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=timeout)
    raw=p.stdout
    if (PRIVATE/'password').exists(): raw=raw.replace((PRIVATE/'password').read_bytes(),b'[REDACTED]')
    return p.returncode,raw,{'argv':args,'cwd':str(cwd),'utc_start':start,'utc_end':utc(),'exit':p.returncode}
mode=sys.argv[1]
if mode=='provision':
    PRIVATE.mkdir(mode=0o700,exist_ok=False)
    password=secrets.token_hex(24)
    (PRIVATE/'password').write_text(password); (PRIVATE/'password').chmod(0o600)
    (PRIVATE/'postgres.env').write_text('POSTGRES_USER=mc_fixture\nPOSTGRES_DB=mc_fixture\nPOSTGRES_PASSWORD='+password+'\n'); (PRIVATE/'postgres.env').chmod(0o600)
    name='mc-codex-be-local-20261029-r2'
    args=['docker','run','-d','--name',name,'--label','mc.owner=MC-P0-CODEX-BACKEND-LOCAL-20261029','--cpus','1','--memory','512m','--pids-limit','128','--publish','127.0.0.1::5432','--env-file',str(PRIVATE/'postgres.env'),'898c60cb390d']
    code,raw,event=run(args); assert code==0,raw.decode()
    (PRIVATE/'container').write_text(name)
    for _ in range(100):
        code,_,_=run(['docker','exec',name,'pg_isready','-U','mc_fixture','-d','mc_fixture'])
        if code==0: break
        time.sleep(.2)
    assert code==0
    code,raw,_=run(['docker','port',name,'5432/tcp']); assert code==0
    port=int(raw.decode().strip().rsplit(':',1)[1])
    dsn=f'postgres://mc_fixture:{password}@127.0.0.1:{port}/mc_fixture?sslmode=disable'
    (PRIVATE/'dsn').write_text(dsn); (PRIVATE/'dsn').chmod(0o600)
    code,raw,_=run(['docker','inspect','--format','{{ index .Config.Labels "mc.owner" }} {{.Image}} {{.HostConfig.Memory}} {{.HostConfig.NanoCpus}} {{.HostConfig.PidsLimit}}',name]); assert code==0
    (OUT/'fixture.json').write_text(json.dumps({'provision':event,'name':name,'binding':f'127.0.0.1:{port}','ownership_and_limits':raw.decode().strip(),'ready':True,'private_directory':str(PRIVATE)},indent=2)+'\n')
    print('Created isolated labelled fixture',name,'127.0.0.1:'+str(port))
elif mode in ('baseline','check'):
    name=(PRIVATE/'container').read_text()
    code,raw,_=run(['docker','inspect','--format','{{ index .Config.Labels "mc.owner" }}',name]); assert code==0 and raw.strip()==b'MC-P0-CODEX-BACKEND-LOCAL-20261029'
    env=os.environ.copy()
    for k in list(env):
        if k.startswith(('MC_TEST_R2_','AWS_','MC_DATABASE','MC_MIGRATE_DATABASE','TEST_DATABASE')): del env[k]
    env['MC_TEST_DATABASE_URL']=(PRIVATE/'dsn').read_text()
    if mode=='baseline': args=['go','test','-json','-race','-count=1','./...']
    else: args=['make','check']; env['GOFLAGS']='-v'
    code,raw,event=run(args,env,ROOT/'apps/backend',1200)
    (OUT/(mode+'.log')).write_bytes(raw)
    (OUT/(mode+'-command.json')).write_text(json.dumps(event,indent=2)+'\n')
    if mode=='baseline':
        counts=collections.Counter(); skips=[]
        for line in raw.decode().splitlines():
            try: e=json.loads(line)
            except ValueError: continue
            if e.get('Test') and e['Action'] in ['pass','fail','skip']:
                counts[e['Action']]+=1
                if e['Action']!='pass': skips.append({k:e[k] for k in ['Package','Test','Action']})
        (OUT/'baseline-counts.json').write_text(json.dumps({'counts':dict(counts),'nonpass':skips},indent=2)+'\n')
        print('baseline',code,dict(counts),skips)
    else:
        counts={k:sum(1 for l in raw.decode().splitlines() if l.lstrip().startswith('--- '+v+':')) for k,v in [('pass','PASS'),('fail','FAIL'),('skip','SKIP')]}
        (OUT/'check-counts.json').write_text(json.dumps(counts,indent=2)+'\n')
        print('make check',code,counts,raw.decode()[-2000:])
    sys.exit(code)
